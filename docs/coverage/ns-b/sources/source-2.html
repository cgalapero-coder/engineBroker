 package com.gtech.automaticbroker.managedatabase.util.observability;
 
 import io.micrometer.core.instrument.MeterRegistry;
 import org.springframework.stereotype.Component;
 
 @Component
 public class ProcessingMetrics {
 
     private final MeterRegistry meterRegistry;
 
     public ProcessingMetrics(MeterRegistry meterRegistry) {
         this.meterRegistry = meterRegistry;
     }
 
     public void consumed() {
         meterRegistry.counter("microservice.kafka.requests", "outcome", "consumed").increment();
     }
 
     public void processed() {
         meterRegistry.counter("microservice.kafka.requests", "outcome", "processed").increment();
     }
 
     public void failed() {
         meterRegistry.counter("microservice.kafka.requests", "outcome", "failed").increment();
     }
 
     public void responsePublished() {
         meterRegistry.counter("microservice.kafka.responses", "outcome", "published").increment();
     }
 }
