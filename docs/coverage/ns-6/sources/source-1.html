 package com.gtech.automaticbroker.managedatabase.kafka.entity;
 
 import com.gtech.automaticbroker.managedatabase.util.exception.InvalidMessageException;
 import java.util.UUID;
 import org.springframework.stereotype.Component;
 import tools.jackson.core.JacksonException;
 import tools.jackson.databind.ObjectMapper;
 
 @Component
 public class KafkaMessageCodec {
 
     private final ObjectMapper objectMapper;
 
     public KafkaMessageCodec(ObjectMapper objectMapper) {
         this.objectMapper = objectMapper;
     }
 
     public RequestEnvelope decodeRequest(String rawMessage) {
         try {
             RequestEnvelope request = objectMapper.readValue(rawMessage, RequestEnvelope.class);
             if (request.messageId() == null || request.messageId().isBlank()
                     || request.correlationId() == null || request.correlationId().isBlank()
                     || request.operation() == null
                     || request.payload() == null || request.payload().isNull()
                     || request.responseRequired() == null) {
                 throw new InvalidMessageException("Request is missing required envelope fields");
             }
             return request;
         } catch (JacksonException exception) {
             throw new InvalidMessageException("Request is not valid JSON", exception);
         }
     }
 
     public String encodeResponse(ResponseEnvelope response) {
         try {
             return objectMapper.writeValueAsString(response);
         } catch (JacksonException exception) {
             throw new IllegalStateException("Response could not be serialized", exception);
         }
     }
 
     public static String newMessageId() {
         return UUID.randomUUID().toString();
     }
 }
