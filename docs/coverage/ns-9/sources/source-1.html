 package com.gtech.automaticbroker.managedatabase.service.impl;
 
 import com.gtech.automaticbroker.managedatabase.domain.entity.CrudDocument;
 import com.gtech.automaticbroker.managedatabase.domain.repository.CrudStore;
 import com.gtech.automaticbroker.managedatabase.service.CrudService;
 import com.gtech.automaticbroker.managedatabase.service.DocumentIdGenerator;
 import com.gtech.automaticbroker.managedatabase.util.exception.DocumentNotFoundException;
 import com.gtech.automaticbroker.managedatabase.util.exception.DuplicateDocumentException;
 import com.gtech.automaticbroker.managedatabase.util.exception.InvalidDocumentException;
 
 import java.lang.reflect.Array;
 import java.lang.reflect.Field;
 import java.util.ArrayList;
 import java.util.Arrays;
 import java.util.LinkedHashMap;
 import java.util.List;
 import java.util.Map;
 import java.util.Set;
 import java.util.stream.Collectors;
 
 import org.springframework.dao.DuplicateKeyException;
 
 public abstract class AbstractCrudServiceImpl<T extends CrudDocument> implements CrudService<T> {
 
     private final CrudStore<T> store;
     private final DocumentIdGenerator idGenerator;
     private final String collectionName;
     private final Set<String> mutableFields;
 
     protected AbstractCrudServiceImpl(
             CrudStore<T> store,
             DocumentIdGenerator idGenerator,
             Class<T> documentType,
             String collectionName) {
         this.store = store;
         this.idGenerator = idGenerator;
         this.collectionName = collectionName;
         this.mutableFields = Arrays.stream(documentType.getDeclaredFields())
                 .map(Field::getName)
                 .collect(Collectors.toUnmodifiableSet());
     }
 
     @Override
     public T create(T document) {
         if (document == null) {
             throw new InvalidDocumentException("Document must not be null");
         }
         validateNoPlaintextPasswords(document);
         if (document.getId() == null || document.getId().isBlank()) {
             document.setId(idGenerator.generate());
         }
         try {
             return removePlaintextPasswords(store.insert(document));
         } catch (DuplicateKeyException exception) {
             throw new DuplicateDocumentException(collectionName);
         }
     }
 
     @Override
     public T getById(String id) {
         validateId(id);
         return store.findById(id)
                 .map(this::removePlaintextPasswords)
                 .orElseThrow(() -> new DocumentNotFoundException(collectionName, id));
     }
 
     @Override
     public List<T> getAll() {
         return store.findAll().stream()
                 .map(this::removePlaintextPasswords)
                 .toList();
     }
 
     @Override
     public T update(String id, Map<String, Object> changes) {
         validateId(id);
         if (changes == null) {
             throw new InvalidDocumentException("Changes must not be null");
         }
         if (changes.isEmpty()) {
             return getById(id);
         }
         changes.keySet().forEach(this::validateMutableField);
         validateNoPlaintextPasswords(changes);
         return store.update(id, changes)
                 .map(this::removePlaintextPasswords)
                 .orElseThrow(() -> new DocumentNotFoundException(collectionName, id));
     }
 
     @Override
     public boolean deleteById(String id) {
         validateId(id);
         return store.deleteById(id);
     }
 
     private void validateId(String id) {
         if (id == null || id.isBlank()) {
             throw new InvalidDocumentException("Document id must not be blank");
         }
     }
 
     private void validateMutableField(String field) {
         if (field == null || field.equals("id") || field.equals("_id") || !mutableFields.contains(field)) {
             throw new InvalidDocumentException("Field is not updatable: " + field);
         }
     }
 
     private void validateNoPlaintextPasswords(Object value) {
         if (value instanceof Map<?, ?> map) {
             for (Map.Entry<?, ?> entry : map.entrySet()) {
                 if (entry.getKey() instanceof String key && key.equalsIgnoreCase("password")) {
                     throw new InvalidDocumentException("Plaintext password fields are not accepted");
                 }
                 validateNoPlaintextPasswords(entry.getValue());
             }
         } else if (value instanceof Iterable<?> values) {
             values.forEach(this::validateNoPlaintextPasswords);
         } else if (value != null && value.getClass().isArray()) {
             for (int index = 0; index < Array.getLength(value); index++) {
                 validateNoPlaintextPasswords(Array.get(value, index));
             }
         } else if (value instanceof CrudDocument document) {
             Arrays.stream(document.getClass().getDeclaredFields())
                     .map(field -> {
                         try {
                             field.setAccessible(true);
                             return field.get(document);
                         } catch (IllegalAccessException exception) {
                             throw new IllegalStateException("Unable to validate document fields", exception);
                         }
                     })
                     .forEach(this::validateNoPlaintextPasswords);
         }
     }
 
     private T removePlaintextPasswords(T document) {
         sanitizeDocumentFields(document);
         return document;
     }
 
     private void sanitizeDocumentFields(CrudDocument document) {
         Arrays.stream(document.getClass().getDeclaredFields()).forEach(field -> {
             try {
                 field.setAccessible(true);
                 field.set(document, sanitizedValue(field.get(document)));
             } catch (IllegalAccessException exception) {
                 throw new IllegalStateException("Unable to sanitize document fields", exception);
             }
         });
     }
 
     private Object sanitizedValue(Object value) {
         if (value instanceof Map<?, ?> map) {
             Map<Object, Object> sanitized = new LinkedHashMap<>();
             map.forEach((key, nestedValue) -> {
                 if (!(key instanceof String stringKey && stringKey.equalsIgnoreCase("password"))) {
                     sanitized.put(key, sanitizedValue(nestedValue));
                 }
             });
             return sanitized;
         }
         if (value instanceof Iterable<?> values) {
             List<Object> sanitized = new ArrayList<>();
             values.forEach(nestedValue -> sanitized.add(sanitizedValue(nestedValue)));
             return sanitized;
         }
         if (value != null && value.getClass().isArray()) {
             List<Object> sanitized = new ArrayList<>();
             for (int index = 0; index < Array.getLength(value); index++) {
                 sanitized.add(sanitizedValue(Array.get(value, index)));
             }
             return sanitized;
         }
         if (value instanceof CrudDocument document) {
             sanitizeDocumentFields(document);
             return document;
         }
         return value;
     }
 }
