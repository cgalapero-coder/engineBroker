 package com.gtech.automaticbroker.managedatabase.kafka.consumer;
 
 import com.gtech.automaticbroker.managedatabase.kafka.entity.KafkaMessageCodec;
 import com.gtech.automaticbroker.managedatabase.kafka.entity.RequestEnvelope;
 import com.gtech.automaticbroker.managedatabase.kafka.entity.ResponseEnvelope;
 import com.gtech.automaticbroker.managedatabase.service.OperationCommand;
 import com.gtech.automaticbroker.managedatabase.service.OperationResult;
 import com.gtech.automaticbroker.managedatabase.service.OperationService;
 import com.gtech.automaticbroker.managedatabase.util.exception.InvalidMessageException;
 import com.gtech.automaticbroker.managedatabase.util.observability.ProcessingMetrics;
 import com.gtech.automaticbroker.managedatabase.kafka.producer.KafkaResponsePublisher;
 import java.time.Instant;
 import org.slf4j.Logger;
 import org.slf4j.LoggerFactory;
 import org.slf4j.MDC;
 import org.springframework.kafka.annotation.KafkaListener;
 import org.springframework.stereotype.Component;
 import tools.jackson.core.JacksonException;
 import tools.jackson.databind.JsonNode;
 import tools.jackson.databind.ObjectMapper;
 
 @Component
 public class KafkaRequestListener {
 
     private static final Logger LOGGER = LoggerFactory.getLogger(KafkaRequestListener.class);
 
     private final KafkaMessageCodec codec;
     private final OperationService operationService;
     private final KafkaResponsePublisher responsePublisher;
     private final ProcessingMetrics metrics;
     private final ObjectMapper objectMapper;
 
     public KafkaRequestListener(
             KafkaMessageCodec codec,
             OperationService operationService,
             KafkaResponsePublisher responsePublisher,
             ProcessingMetrics metrics,
             ObjectMapper objectMapper) {
         this.codec = codec;
         this.operationService = operationService;
         this.responsePublisher = responsePublisher;
         this.metrics = metrics;
         this.objectMapper = objectMapper;
     }
 
     @KafkaListener(topics = "${app.messaging.request-topic}")
     public void receive(String rawMessage) {
         metrics.consumed();
         RequestEnvelope request = null;
         try {
             request = codec.decodeRequest(rawMessage);
             MDC.put("correlationId", request.correlationId());
             OperationResult result = operationService.execute(
                     new OperationCommand(request.operation(), toJson(request.payload())));
             if (request.responseRequired()) {
                 responsePublisher.publish(toResponse(request, result));
                 metrics.responsePublished();
             }
             metrics.processed();
             LOGGER.info("Kafka request processed: operation={}, responseRequired={}",
                     request.operation(), request.responseRequired());
         } catch (RuntimeException exception) {
             metrics.failed();
             LOGGER.error("Kafka request processing failed: operation={}",
                     request == null ? "unknown" : request.operation(), exception);
             throw exception;
         } finally {
             MDC.remove("correlationId");
         }
     }
 
     private String toJson(JsonNode payload) {
         try {
             return objectMapper.writeValueAsString(payload);
         } catch (JacksonException exception) {
             throw new InvalidMessageException("Request payload could not be serialized", exception);
         }
     }
 
     private ResponseEnvelope toResponse(RequestEnvelope request, OperationResult result) {
         JsonNode responsePayload = parseResult(result.resultJson());
         ResponseEnvelope.ResponseError error = result.errorCode() == null
                 ? null
                 : new ResponseEnvelope.ResponseError(
                         result.errorCategory(), result.errorCode(), result.errorMessage());
 
         return new ResponseEnvelope(
                 KafkaMessageCodec.newMessageId(),
                 request.messageId(),
                 request.correlationId(),
                 request.operation(),
                 result.status(),
                 responsePayload,
                 error,
                 Instant.now());
     }
 
     private JsonNode parseResult(String resultJson) {
         if (resultJson == null) {
             return null;
         }
         try {
             return objectMapper.readTree(resultJson);
         } catch (JacksonException exception) {
             throw new IllegalStateException("Operation result is not valid JSON", exception);
         }
     }
 }
